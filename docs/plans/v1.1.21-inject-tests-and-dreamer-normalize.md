## Design — Graph Normalize Phase

### Detection rules (read-only first pass)

**Entity-similarity candidates:** for every pair of entities `(a, b)` where `a.id < b.id`:
1. Same `entity_type`.
2. Same slug prefix — split on first `_`, prefixes must match (`person_tima` vs `person_timofey` → both `person`).
3. Levenshtein **ratio** on the suffix portion (post-prefix), where ratio = `1 - lev(a, b) / max(len(a), len(b))`.
4. Suffix length ≥ 2 on both sides.

**Relation-canonical map** (hardcoded in v1.1.21):
```
references → mentions
refers_to  → mentions
contributes_to → works_on
at_location → located_at
```

**Orphan candidates:** entities with zero in+out **active** relations (where `valid_to IS NULL`), `entity_type NOT IN ('kind','area','agent','workspace')`, age > 14 days.

**Broken relations:** `relations.valid_to IS NULL` rows where `from_id` or `to_id` does not exist in `entities`.

### Action policy

**Auto-merge** entity pair `(a, b)` ONLY IF:
- LD-ratio ≥ 0.92 AND
- one side has strictly fewer relations than the other (asymmetric).
- Lower-degree side becomes alias; canonical = higher-degree.
- Use existing `graph::add_alias(conn, alias_id, canonical_id)`.

**Flag** entity pair `(a, b)` IF:
- LD-ratio in [0.85, 0.92) OR balanced relation degrees.
- Recorded in dream report stdout JSON under `flagged_entity_pairs[]`.

**Synonymous relations:** `UPDATE relations SET relation_type = canonical WHERE relation_type IN (synonyms) AND valid_to IS NULL`. UNIQUE constraint may collide on `(from_id, relation_type, to_id, valid_from)`; on collision fall back to deleting the synonym row (canonical already exists).

**Orphans:** `DELETE FROM entities WHERE id = ?` for each match.

**Broken relations:** `UPDATE relations SET valid_to = ? WHERE id = ? AND valid_to IS NULL`.

### Report shape

```json
{
  "ok": true,
  "report": {
    "phase": "normalize",
    "merged": 3,
    "flagged_entity_pairs": [
      {"a": "person_anya", "b": "person_anna", "ratio": 0.875}
    ],
    "relations_canonicalized": 7,
    "orphans_deleted": 12,
    "broken_relations_invalidated": 2
  }
}
```

```python
Pins the contract that emerged from v1.1.18-v1.1.20:
  - ok=true always (Enter always fires, even on verify timeout).
  - The Enter byte is actually written (recipient sees a fresh prompt).
  - The 'verified' field reflects observed evidence, not a hardcoded true.
"""
from __future__ import annotations

import json
import os
import string
import time
import uuid
from typing import Any

import pytest
import requests

BASE = os.environ.get("CODETERM_API_BASE", "http://127.0.0.1:7686")
TOKEN = os.environ.get("CODETERM_API_TOKEN", "")
HEADERS = {"Authorization": f"Bearer {TOKEN}"} if TOKEN else {}


def _post(path: str, body: dict[str, Any]) -> dict[str, Any]:
    r = requests.post(f"{BASE}{path}", json=body, headers=HEADERS, timeout=30)
    r.raise_for_status()
    return r.json()


def _get(path: str) -> dict[str, Any]:
    r = requests.get(f"{BASE}{path}", headers=HEADERS, timeout=30)
    r.raise_for_status()
    return r.json()


def _alphanumeric(n: int) -> str:
    pool = string.ascii_letters + string.digits
    rng = uuid.uuid4().int
    out = []
    for _ in range(n):
        rng, idx = divmod(rng, len(pool))
        if rng == 0:
            rng = uuid.uuid4().int
        out.append(pool[idx])
    return "".join(out)


@pytest.fixture(scope="module")
def shell_pane() -> str:
    """Spawn a plain shell pane in a throwaway workspace, return pane id.

    The verifier observes screen state, not Claude-CLI rendering, so a
    bash/pwsh prompt is sufficient. A fresh prompt after our marker
    proves the Enter byte landed in the PTY.
    """
    ws = _post("/api/workspaces", {"name": f"inject-test-{uuid.uuid4().hex[:8]}"})
    ws_id = ws["id"]
    pane = _post(f"/api/workspaces/{ws_id}/panes", {"kind": "shell"})
    pane_id = pane["id"]
    # Give the shell a beat to render its first prompt.
    time.sleep(0.5)
    yield pane_id
    # Best-effort cleanup; ignore failures.
    try:
        requests.delete(f"{BASE}/api/workspaces/{ws_id}", headers=HEADERS, timeout=10)
    except Exception:
        pass


def _send_and_check(pane_id: str, text: str, marker: str) -> dict[str, Any]:
    """Send text+Enter, return the parsed response. Caller asserts."""
    return _post(f"/api/panes/{pane_id}/send", {"text": text, "enter": True})


def _output_after(pane_id: str, deadline_s: float = 4.0) -> str:
    """Poll until `deadline_s` for the latest pane output."""
    end = time.time() + deadline_s
    last = ""
    while time.time() < end:
        out = _get(f"/api/panes/{pane_id}/output?strip_ansi=true&full=true")
        last = out.get("text", "") or out.get("output", "")
        time.sleep(0.2)
    return last


# --- the 4 size cases ---------------------------------------------------

CASES = [
    ("short40", lambda: f"echo short_marker_{_alphanumeric(20)}"),
    ("medium700", lambda: f"echo {_alphanumeric(700)}"),
    ("single5k", lambda: f"echo {_alphanumeric(5000)}"),
    # Multi-line via printf '\n' embedding so the renderer wraps onto many rows.
    ("multi5k", lambda: "printf '" + ("MULTI_" + _alphanumeric(60) + "\\n") * 70 + "'"),
]


@pytest.mark.parametrize("label,builder", CASES, ids=[c[0] for c in CASES])
def test_inject_verify_contract(shell_pane: str, label: str, builder) -> None:
    text = builder()
    marker = text.split()[1][:30] if " " in text else text[:30]

    resp = _send_and_check(shell_pane, text, marker)

    # 1. ok=true ALWAYS — Enter must always fire.
    assert resp.get("ok") is True, f"{label}: ok must be true; got {resp}"

    # 2. verified field MUST reflect observed evidence (echo + submit), not be
    #    hardcoded. Either both observed -> True, or one missing -> False.
    echo_observed = bool(resp.get("echo_observed", False))
    submit_observed = bool(resp.get("submit_observed", False))
    expected_verified = echo_observed and submit_observed
    assert resp.get("verified") == expected_verified, (
        f"{label}: verified flag is hardcoded — got {resp.get('verified')!r}, "
        f"expected {expected_verified} (echo={echo_observed}, submit={submit_observed})"
    )

    # 3. The Enter byte was actually written: recipient sees the marker AND a
    #    fresh prompt follows it. We check the marker for short payloads only;
    #    long ones may have scrolled out of the visible screen state.
    output = _output_after(shell_pane)
    if label == "short40":
        assert marker in output, f"{label}: marker {marker!r} not in output: {output[-500:]!r}"
        # A fresh prompt after the marker proves Enter fired in the PTY.
        tail = output.rsplit(marker, 1)[-1]
        # Bash $ or PowerShell PS> — at least one prompt sigil after the marker.
        assert any(s in tail for s in ("$", ">", "#")), (
            f"{label}: no fresh prompt after marker; tail={tail[-200:]!r}"
        )

    # 4. For verify-timeout cases (long payloads) ok must still be true. Already
    #    asserted above; we re-pin to make the regression intent obvious.
    assert resp["ok"] is True, f"{label}: ok must remain true on verify timeout"


def test_inject_verify_short_is_fully_verified(shell_pane: str) -> None:
    """Short clean payloads MUST be fully verified — if this regresses, the
    verifier itself broke (not just a long-payload edge case)."""
    text = "echo verifyok_" + _alphanumeric(10)
    resp = _send_and_check(shell_pane, text, text.split()[1])
    assert resp["ok"] is True
    assert resp["echo_observed"] is True, f"short payload echo must be observed: {resp}"
    assert resp["submit_observed"] is True, f"short payload submit must be observed: {resp}"
    assert resp["verified"] is True, f"short payload must be verified: {resp}"
```

```rust
#[derive(Debug, Clone, Default)]
struct WorkerNormalizeReport {
    merged: u64,
    flagged_pairs: Vec<serde_json::Value>,
    relations_canonicalized: u64,
    orphans_deleted: u64,
    broken_relations_invalidated: u64,
}

const PROTECTED_ENTITY_TYPES: &[&str] = &["kind", "area", "agent", "workspace"];
const ORPHAN_AGE_DAYS: i64 = 14;
const MERGE_RATIO_HARD: f64 = 0.92;
const MERGE_RATIO_FLAG: f64 = 0.85;

fn canonical_relation_type(rel: &str) -> Option<&'static str> {
    match rel {
        "references" | "refers_to" => Some("mentions"),
        "contributes_to" => Some("works_on"),
        "at_location" => Some("located_at"),
        _ => None,
    }
}

fn levenshtein(a: &str, b: &str) -> usize {
    let a: Vec<char> = a.chars().collect();
    let b: Vec<char> = b.chars().collect();
    let (n, m) = (a.len(), b.len());
    if n == 0 { return m; }
    if m == 0 { return n; }
    let mut prev: Vec<usize> = (0..=m).collect();
    let mut curr = vec![0usize; m + 1];
    for i in 1..=n {
        curr[0] = i;
        for j in 1..=m {
            let cost = if a[i - 1] == b[j - 1] { 0 } else { 1 };
            curr[j] = (prev[j] + 1).min(curr[j - 1] + 1).min(prev[j - 1] + cost);
        }
        std::mem::swap(&mut prev, &mut curr);
    }
    prev[m]
}

fn lev_ratio(a: &str, b: &str) -> f64 {
    let max = a.chars().count().max(b.chars().count());
    if max == 0 { return 1.0; }
    1.0 - (levenshtein(a, b) as f64 / max as f64)
}

fn split_prefix(id: &str) -> Option<(&str, &str)> {
    id.find('_').map(|i| (&id[..i], &id[i + 1..]))
}

fn worker_dreaming_normalize(conn: &rusqlite::Connection) -> Result<serde_json::Value, String> {
    let mut report = WorkerNormalizeReport::default();
    let now = chrono::Utc::now().to_rfc3339();

    // 1. Canonicalize synonymous relations.
    let synonyms: &[(&str, &str)] = &[
        ("references", "mentions"),
        ("refers_to", "mentions"),
        ("contributes_to", "works_on"),
        ("at_location", "located_at"),
    ];
    for (from, to) in synonyms {
        // Try in-place rename; on UNIQUE collision, drop the synonym row.
        let updated = conn.execute(
            "UPDATE OR IGNORE relations SET relation_type = ?1 WHERE relation_type = ?2 AND valid_to IS NULL",
            rusqlite::params![to, from],
        ).map_err(|e| format!("canonicalize update: {}", e))? as u64;
        let dropped = conn.execute(
            "DELETE FROM relations WHERE relation_type = ?1 AND valid_to IS NULL",
            rusqlite::params![from],
        ).map_err(|e| format!("canonicalize delete: {}", e))? as u64;
        report.relations_canonicalized += updated + dropped;
    }

    // 2. Detect entity merge candidates.
    let entities: Vec<(String, String)> = {
        let mut stmt = conn
            .prepare("SELECT id, entity_type FROM entities ORDER BY id")
            .map_err(|e| e.to_string())?;
        stmt.query_map([], |r| Ok((r.get::<_, String>(0)?, r.get::<_, String>(1)?)))
            .map_err(|e| e.to_string())?
            .filter_map(|r| r.ok())
            .collect()
    };

    // Pre-fetch active relation degrees per entity.
    let degrees: std::collections::HashMap<String, u32> = {
        let mut stmt = conn
            .prepare(
                "SELECT id, (
                    SELECT COUNT(*) FROM relations
                    WHERE valid_to IS NULL AND (from_id = entities.id OR to_id = entities.id)
                 ) FROM entities",
            )
            .map_err(|e| e.to_string())?;
        stmt.query_map([], |r| Ok((r.get::<_, String>(0)?, r.get::<_, u32>(1)?)))
            .map_err(|e| e.to_string())?
            .filter_map(|r| r.ok())
            .collect()
    };

    // Group by (entity_type, prefix) to keep pairwise checks cheap.
    use std::collections::HashMap;
    let mut buckets: HashMap<(String, String), Vec<(String, String)>> = HashMap::new();
    for (id, etype) in &entities {
        if let Some((prefix, suffix)) = split_prefix(id) {
            if suffix.chars().count() >= 2 {
                buckets
                    .entry((etype.clone(), prefix.to_string()))
                    .or_default()
                    .push((id.clone(), suffix.to_string()));
            }
        }
    }

    for entries in buckets.values() {
        for i in 0..entries.len() {
            for j in (i + 1)..entries.len() {
                let (a_id, a_suf) = &entries[i];
                let (b_id, b_suf) = &entries[j];
                let ratio = lev_ratio(a_suf, b_suf);
                if ratio < MERGE_RATIO_FLAG {
                    continue;
                }
                let a_deg = degrees.get(a_id).copied().unwrap_or(0);
                let b_deg = degrees.get(b_id).copied().unwrap_or(0);
                if ratio >= MERGE_RATIO_HARD && a_deg != b_deg {
                    let (alias, canonical) = if a_deg < b_deg {
                        (a_id.clone(), b_id.clone())
                    } else {
                        (b_id.clone(), a_id.clone())
                    };
                    if codeterm_lib::mem::graph_store::add_alias(conn, &alias, &canonical).is_ok() {
                        report.merged += 1;
                    }
                } else {
                    report.flagged_pairs.push(serde_json::json!({
                        "a": a_id, "b": b_id, "ratio": ratio,
                    }));
                }
            }
        }
    }

    // 3. Orphan cleanup.
    let cutoff = (chrono::Utc::now() - chrono::Duration::days(ORPHAN_AGE_DAYS)).to_rfc3339();
    let placeholders = PROTECTED_ENTITY_TYPES
        .iter()
        .map(|_| "?")
        .collect::<Vec<_>>()
        .join(",");
    let sql = format!(
        "DELETE FROM entities WHERE entity_type NOT IN ({}) \
         AND created_at < ?{} \
         AND id NOT IN (SELECT from_id FROM relations WHERE valid_to IS NULL) \
         AND id NOT IN (SELECT to_id FROM relations WHERE valid_to IS NULL)",
        placeholders,
        PROTECTED_ENTITY_TYPES.len() + 1,
    );
    let mut params: Vec<&dyn rusqlite::ToSql> =
        PROTECTED_ENTITY_TYPES.iter().map(|s| s as &dyn rusqlite::ToSql).collect();
    params.push(&cutoff);
    let n = conn
        .execute(&sql, params.as_slice())
        .map_err(|e| format!("orphan cleanup: {}", e))? as u64;
    report.orphans_deleted = n;

    // 4. Broken-relation invalidation.
    let broken = conn
        .execute(
            "UPDATE relations SET valid_to = ?1 \
             WHERE valid_to IS NULL \
             AND (from_id NOT IN (SELECT id FROM entities) \
                  OR to_id NOT IN (SELECT id FROM entities))",
            rusqlite::params![&now],
        )
        .map_err(|e| format!("broken invalidate: {}", e))? as u64;
    report.broken_relations_invalidated = broken;

    Ok(serde_json::json!({
        "phase": "normalize",
        "merged": report.merged,
        "flagged_entity_pairs": report.flagged_pairs,
        "relations_canonicalized": report.relations_canonicalized,
        "orphans_deleted": report.orphans_deleted,
        "broken_relations_invalidated": report.broken_relations_invalidated,
    }))
}
```

```rust
#[derive(Deserialize, Default)]
pub struct DreamParams {
    #[serde(default)]
    pub phase: Option<String>,
}

pub async fn mem_dream(
    AxumState(state): AxumState<AppState>,
    headers: HeaderMap,
    Query(p): Query<DreamParams>,
) -> Result<Json<serde_json::Value>, (StatusCode, String)> {
    check_auth(&state, &headers).map_err(|(s, e)| (s, e.to_string()))?;
    use crate::mem::supervisor::WorkerKind;

    let phase = p.phase.as_deref().unwrap_or("all").to_lowercase();
    let phases: Vec<WorkerKind> = match phase.as_str() {
        "light" => vec![WorkerKind::DreamingLight],
        "deep" => vec![WorkerKind::DreamingDeep],
        "normalize" => vec![WorkerKind::DreamingNormalize],
        "all" => vec![WorkerKind::DreamingLight, WorkerKind::DreamingDeep, WorkerKind::DreamingNormalize],
        other => return Err((StatusCode::BAD_REQUEST, format!("unknown phase: {}", other))),
    };

    let mut reports = Vec::new();
    for k in &phases {
        let r = state.svc.mem_supervisor.run(*k).await
            .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("{:?}: {}", k, e)))?;
        reports.push(r);
    }

    // REM worker only on full nightly run, not for targeted phase calls.
    if phase == "all" {
        let svc = state.svc.clone();
        std::thread::spawn(move || dreaming::spawn_rem_worker(&svc));
    }

    Ok(Json(serde_json::json!({ "phases": reports })))
}
```
